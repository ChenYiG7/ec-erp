CREATE TABLE product (

## 4. 仓库与库存

```sql
CREATE TABLE warehouse (
  id      BIGINT AUTO_INCREMENT PRIMARY KEY,
  wh_name VARCHAR(128) NOT NULL,
  wh_type VARCHAR(32) NOT NULL DEFAULT 'SELF' COMMENT 'SELF自仓/FBA/OVERSEAS海外仓/VIRTUAL虚拟仓',
  country VARCHAR(8),
  address VARCHAR(512),
  status  TINYINT NOT NULL DEFAULT 1
) COMMENT '仓库';

CREATE TABLE inventory (
  id            BIGINT AUTO_INCREMENT PRIMARY KEY,
  sku_id        BIGINT NOT NULL,
  warehouse_id  BIGINT NOT NULL,
  qty_on_hand   INT NOT NULL DEFAULT 0 COMMENT '在库',
  qty_locked    INT NOT NULL DEFAULT 0 COMMENT '占用(发货单占用未发货:建单占用/取消·删除·改单释放,#11)',
  qty_transit   INT NOT NULL DEFAULT 0 COMMENT '在途(采购审核占用未入库:审核占/关闭释放,#10)',
  qty_available INT NOT NULL DEFAULT 0 COMMENT '可用=在库-占用,由 InventoryService 同事务维护,禁止旁路 update',
  UNIQUE KEY uk_sku_wh (sku_id, warehouse_id)
) COMMENT '库存(分仓,只能经 inventory_flow 变动)';
-- SKU 级安全库存(safety_stock)二期随"安全库存预警"功能加列

CREATE TABLE inventory_flow (
  id           BIGINT AUTO_INCREMENT PRIMARY KEY,
  sku_id       BIGINT NOT NULL,
  warehouse_id BIGINT NOT NULL,
  flow_type    VARCHAR(32) NOT NULL COMMENT 'IN_PURCHASE/OUT_SHIP/IN_RETURN/ADJUST/TRANSFER_OUT/TRANSFER_IN/IN_TRANSIT/LOCK_SHIP/IN_TRANSFER(#7 2026-09-06 八值,2026-09-12 #30 余量① 增 IN_TRANSFER 调拨到货核销成九值)',
  quantity     INT NOT NULL COMMENT '正负数',
  before_qty   INT NOT NULL, after_qty INT NOT NULL,
  biz_type     VARCHAR(32) COMMENT '关联业务类型',
  biz_id       BIGINT COMMENT '关联单据ID',
  unit_cost    DECIMAL(18,8) NULL COMMENT '动账单价快照(CNY,移动加权 #19③)',
  cost_amount  DECIMAL(18,2) NULL COMMENT '动账成本额(CNY,带符号)',
  created_by   BIGINT,
  KEY idx_sku_time (sku_id, created_at)
) COMMENT '库存流水(只增不改,与库存变更同事务写入)';

-- SKU 移动加权成本账(#19③ 利润核算 V1 2026-09-08 定稿,docs/02 §14 成本计价拍板"先移动加权")
CREATE TABLE sku_cost_state (
  id           BIGINT AUTO_INCREMENT PRIMARY KEY,
  sku_id       BIGINT NOT NULL,
  total_qty    INT NOT NULL DEFAULT 0 COMMENT '账本结存数量(全局跨仓,加权分母)',
  total_amount DECIMAL(18,2) NOT NULL DEFAULT 0 COMMENT '账本结存金额(CNY,加权分子)',
  avg_cost     DECIMAL(18,8) NOT NULL DEFAULT 0 COMMENT '当前移动加权单价(CNY);清零保留末次价作暂估基准',
  updated_at   DATETIME,
  UNIQUE KEY uk_sku (sku_id)
) COMMENT 'SKU移动加权成本账(出库成本事实源:OUT_SHIP 行 unit_cost 快照落 inventory_flow)';
```

> 库存变更必须走统一入口(`InventoryService.change`,TODO.md #7),同事务写 flow,禁止直接 update 数量列。
> flow_type 列语义矩阵(#7 2026-09-06 差异化落地,收口 InventoryService.FlowOps,词表三方同步:
> InventoryConsts / 本文档 / 01_schema_init.sql):
>
> | flow_type | 业务动作 | qty_on_hand | qty_locked | qty_transit | qty_available | 守卫(WHERE) |
> |---|---|---|---|---|---|---|
> | IN_TRANSIT | 采购审核占在途(+)/关闭释放(-)/调拨确认占(+#30 余量①) | – | – | +q | – | 无 |
> | IN_PURCHASE | 入库核销(转可用) | +q | – | -q | +q | 在途 ≥ q |
> | IN_TRANSFER | 调拨到货核销(#30 余量① 2026-09-12,列语义同 IN_PURCHASE,biz_type=TRANSFER_ORDER) | +q | – | -q | +q | 在途 ≥ q |
> | IN_RETURN | 售后退货入库 | +q | – | – | +q | 可用+q ≥ 0 |
> | ADJUST | 人工调整(±) | +q | – | – | +q | 可用+q ≥ 0 |
> | TRANSFER_OUT/IN | 调拨两腿(InventoryService.transfer 组合) | +q | – | – | +q | 可用+q ≥ 0 |
> | LOCK_SHIP | 发货建单占用(+)/取消·删除·改单释放(-) | – | +q | – | -q | 可用-q ≥ 0 |
> | OUT_SHIP | 发货出库(核销占用,q<0) | +q | +q | – | –(不变) | 在库/占用 ≥ \|q\| |
>
> 不变量 qty_available = qty_on_hand − qty_locked 全类型成立;IN_TRANSIT/OUT_SHIP 流水 before=after(不动可用)属正常。
> 成本账口径(#19③ 2026-09-08 定稿):`sku_cost_state` 全局跨仓移动加权,动账同事务维护(铁律 4 唯一入口 change,
> 出库结转 SELECT FOR UPDATE state 行串行化,锁序 inventory 行 → state 行单向无死锁)。进账矩阵:
> IN_PURCHASE 进(采购单价,缺价按当时加权暂估禁猜价)/OUT_SHIP 结转(当时加权价,unit_cost 快照落流水)/

### 6.1 利润核算 V1(#19③ 2026-09-08,实时销售利润先行,三口径第一层 docs/02 §14)

- **口径**:订单行粒度,利润 = 售价(CNY) − 出库成本(CNY) − 平台佣金(CNY);已支付态三态(WAIT_SHIP/SHIPPED/COMPLETED)
  同销量日表口径;**不落库实时算**(订单面小时级新鲜度天然满足,V2 周期口径才落库校准)。
- **售价折算**:item_amount × 汇率,汇率按**下单日(so.order_time)回溯 exchange_rate** 取 quoted_at<=order_time 最近一条
  (禁取表内最新;CNY 短路=1;无报价→本位币列为 NULL,禁猜);shop_order.exchange_rate 落库快照仅展示,利润面统一回溯口径。
- **成本**:OUT_SHIP 行 unit_cost 快照(移动加权,见 §4 sku_cost_state),经 delivery_order_item(biz_id=delivery_id)
  按 order_item_id 聚合;未出库行成本 NULL(前端"未发货")。
- **佣金**:settlement_detail fee_type=COMMISSION 按 shop_id+order_item_id(=platform_order_item_id)聚合;
  无结算数据(报告未拉/行无平台行号)→ NULL(前端"待结算");2026-09-11 起缺口可按 `platform_fee_rate`
  **费率估算**(行级 ESTIMATED 标志,见 §6.4),无费率仍为 NULL 禁猜;结算回后以实际值覆盖,差值落周期校差。

定稿偏差(相对首轮草案):结算单 `period` 拆 `period_start/period_end` 两列、补 `settlement_id` 幂等键
(Amazon SettlementId,同店铺唯一,重拉 upsert 防重复入库);明细补 `shop_id` 冗余(同事务写入免联查)、
`order_item_id`(SKU 级利润归集键,对应 shop_order_item.platform_order_item_id)与 `sku` 平台原文列
(本地映射走 #5 绑定关系联查,解析器禁猜);金额全带符号存报告原值(Σ明细=报告头 TotalAmount 即勾稽平,

### 6.2 退款勾稽(#19④ 2026-09-08,#12 遗留"退款金额与财务勾稽"收口)

- **口径**:店铺 + 平台订单号聚合比对(结算 REFUND 行无退款单 ID,逐单勾稽不可行;售后侧经 shop_order
  翻译 platform_order_id,同订单多售后单/部分退款自然 SUM)。售后侧 = 已退款终态(REFUNDED/COMPLETED)
  且 refund_amount 非空;结算侧 = PARSED 报告 REFUND 行(FAILED 明细是待校准暂存态不入比对)。
  售后恒正(绝对值口径)/结算 Σ(−amount) 转正同向。
- **差异判定三类**:`AMOUNT_MISMATCH`(双侧有行且 |差| > 0.01,容差=结算列精度防尾差误报)/
  `MISSING_IN_SETTLEMENT`(售后已退款而 PARSED 报告无该订单 REFUND 行)/`CURRENCY_MISMATCH`
  (任一侧同订单多币种或两侧币种不同,禁混币计算)。勾稽平(|差| ≤ 容差)不告警。
- **落位**:erp-finance RefundReconciliationService 纯读侧(零 DDL,跨域只读 join 同 ProfitQueryMapper
  先例),erp-api RefundReconciliationJob 每日一扫 → 静默期去重 → #14 站内通知扇出
  (notify_type=REFUND_DIFF,聚合单条明细列 topN,模式同 #6 预警)。开关 erp.finance.refund-reconciliation.*。
- **已知边界**(真凭证实测后评估):结算 REFUND 行按 #19② 拍板含退款负佣金行,售后 refund_amount 为
  Principal 本金口径,两侧或存佣金级系统性偏差——差异告警带双侧金额供人工判读,translator 归一
  校准时随 #19② 一并定版。

  **不联动跨仓动账**(海外仓/FBA 库存是否 track 随 fba-shipment 联动拍板)。
- **单据模型四表**:`first_leg_shipment` 头程发货单(状态机 `DRAFT→BOXED→SHIPPED→ALLOCATED→CLOSED`,
  CANCELED 旁路仅 DRAFT/BOXED 可达;国内 SELF 仓→OVERSEAS/FBA 目的仓,经 WarehouseApi 契约校验仓型)、
  `first_leg_box` 箱 + `first_leg_box_item` 箱内件(装箱单数据面;箱记录毛重/外箱长宽高 cm,
  但箱毛重 V1 不进分摊算法——混装箱无规范拆分规则)、`first_leg_alloc` 分摊结果
  (uk(shipment_id,sku_id),**确认后不可重算覆盖,重算=作废重开**)。
- **删除惯例**:单据域物理删除(docs/07 §6.4,仅 DRAFT/CANCELED 可删,箱/内件随主单先删后插);
  SHIPPED 起禁删(运费/分摊已是财务事实);`first_leg_alloc` append-only 不提供删除口。
- **运费与汇率**:SHIPPED 动作录运费(freight_amount 原币 + currency),汇率手填优先、空则按 shipped_at
  回溯 `resolveRate`;非 CNY 无报价**拦截禁猜**,补录汇率快照或手填后才可发货;`freight_cny` 为分摊基准。
- **分摊三策略**(allocate_strategy,默认 WEIGHT):QTY 按跨箱件数;WEIGHT 按 qty×`product_sku.weight_g`;
  AMOUNT 按 qty×最近采购价(`PurchaseQueryApi.findLatestSupplierBySkuIds`,回退 `product_sku.cost_price`)。
  基数全 0 → 降级 QTY 并写 `alloc_remark`(不静默);部分 SKU 缺基数(重量/价格未维护)→ 拦截并列明 SKU;
  `Σ alloc_amount == freight_cny`(尾差并入最大基数行,容差 0.01,不平整不落库)。
- **查询面**:发货单分页/详情(箱+内件+分摊行)外,交付 SKU 维度头程费用汇总(Σ ALLOCATED/CLOSED 单分摊),
  作为后续第三层财务利润报表聚合源;本期不接利润报表(前置第二层 #32 已 2026-09-12 收口,第三层报表未立项随真实使用评估)。

### 6.6 FBA 发货单(V1 内部数据面,2026-09-12,计划书 docs/plans/fba-shipment.md)

- **定位**:FBA 履约链路的内部单据域——发货计划→装箱→发出(国内仓出库动账)→平台收货登记→对账;
  零平台凭证依赖。SP-API Fulfillment Inbound 集成只留 TODO(#35) 槽位(随 #3 真凭证),V1 手工数据面。
  FBA 库存全量跟踪(FBA 仓建 inventory 行)明确延后,平台收货只进对账面。
- **单据模型五表**:`fba_shipment` 发货单(状态机 `DRAFT→BOXED→SHIPPED→RECEIVING→CLOSED`,
  CANCELED 旁路仅 DRAFT/BOXED;国内 SELF 发货仓经 WarehouseApi 契约校验;shop_id/marketplace 为归属信息,
  V1 不做店铺存在性校验——契约无单店视图方法,前端下拉选择兜底)、
  `fba_shipment_item` 计划行(shipment_id+sku_id 唯一;**相对计划书 DDL 草案新增的表**——验收要求的
  「Σbox_item=计划量」装箱勾稽必须有计划量存储位,漂移已记计划书头部)、
  `fba_box` 箱 + `fba_box_item` 箱内件(装箱数据面,同 first_leg 形态)、
  `fba_shipment_diff` 收货对账差异(uk(shipment_id,sku_id);SHIPPED=Σ箱内件发出量 vs 平台收货登记量,
  SHORT 缺收/EXTRA 多收/OK 一致;RECEIVING 态可重复登记覆盖=先删后插,CLOSED 冻结)。
- **库存语义(预拍板 2026-09-10)**:SHIPPED 一步 OUT_SHIP(装箱不占库存,无 LOCK_SHIP——FBA 计划期长,
  长占恶化可用),逐 SKU 经契约 `InventoryChangeApi.change()`(biz_type=`FBA_SHIPMENT`,biz_id=发货单 id),
  成本随货走移动加权成本账(unit_cost 传空由成本账取加权价);可用不足 change() 抛错整单回滚。
  SHIPPED 复合事务含装箱勾稽:Σbox_item 必须逐 SKU 等于计划量,不平时拦截(装箱动作 box() 时先做同款
  预检给早期反馈,SHIPPED 时权威复检)。
- **删除惯例**:单据域物理删除(docs/07 §6.4,仅 DRAFT/CANCELED,计划行/箱/内件随主单先删后插);
  SHIPPED 起库存已动账禁删禁取消(逆向=作废重开,随实际使用拍板);diff 行随登记覆盖,不单独提供删除口。
- **对账 Job**:`FbaReconciliationJob`(erp-api)每日扫 SHIPPED/RECEIVING 超期未关闭的单子聚合一告警
  扇出(#14 三渠道),静默期去重同 RefundReconciliationJob;**默认关**(`erp.fulfill.fba-reconciliation.enabled=false`,
  验收后按需开启);自动取数(报告/API)随 V2 TODO(#35)。

## 设计要点

1. **幂等**:所有平台数据表以 `(shop_id, platform_xxx_id)` 唯一键兜底,拉单可安全重试;shop 表以 `(platform, seller_id)` 防重复建店;
